cmake_minimum_required(VERSION 3.21)
project(liquid-glass-cpp VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC OFF)
set(CMAKE_AUTOUIC OFF)
set(CMAKE_AUTORCC OFF)
add_library(liquid_glass STATIC src/window.cpp)
add_library(liquid_glass::liquid_glass ALIAS liquid_glass)
target_include_directories(liquid_glass PUBLIC include)
target_compile_definitions(liquid_glass PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_definitions(liquid_glass PRIVATE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_link_libraries(liquid_glass PRIVATE d3d11 dxgi d3dcompiler d2d1 dcomp dwmapi user32)
if(MSVC)
    target_compile_options(liquid_glass PRIVATE /utf-8 /W4)
endif()
option(LIQUID_GLASS_BUILD_TESTS "Build independent render tests" ON)
if(LIQUID_GLASS_BUILD_TESTS)
    enable_testing()
    add_executable(liquid_glass_render_tests tests/render_tests.cpp)
    target_link_libraries(liquid_glass_render_tests PRIVATE liquid_glass::liquid_glass)
    add_test(NAME liquid_glass_render COMMAND liquid_glass_render_tests)
    set_tests_properties(liquid_glass_render PROPERTIES LABELS "ui;gpu" TIMEOUT 60)
endif()
